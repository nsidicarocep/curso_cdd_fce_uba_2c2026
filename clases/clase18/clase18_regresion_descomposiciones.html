# Muestra una tabla de modelsummary como tabla markdown dentro del notebook.
# Recibe los mismos argumentos que modelsummary() y, opcionalmente, notas al pie.
mostrar_tabla <- function(modelos, ..., notas = NULL) {
  df <- modelsummary(modelos, ..., output = "data.frame")
  df$term[df$statistic == "std.error"] <- ""
  df <- df[, c("term", names(modelos))]
  names(df)[1] <- " "
  tabla <- paste(knitr::kable(df, format = "pipe",
                              align = c("l", rep("r", length(modelos)))),
                 collapse = "\n")
  notas <- gsub("*", "\\*", notas, fixed = TRUE)
  texto <- paste0(tabla, "\n\n", paste(notas, collapse = "  \n"))
  if (isTRUE(getOption("jupyter.in_kernel"))) IRdisplay::display_markdown(texto) else cat(texto)
}

# Una fila por linea de pedido. El archivo original trae filas basura al final,

                region       = Region,

                categoria    = Category,
                subcategoria = `Sub-Category`,
                ventas       = Sales,
                cantidad     = Quantity,
                descuento    = Discount,
                ganancia     = Profit) |>
      write_csv(local)
  }
  read_csv(local, show_col_types = FALSE)
}


# ---- Properati ---------------------------------------------------------------
# Se baja el archivo completo (con las descripciones de los avisos), se quedan
# solo los departamentos en venta en dolares y se guarda una version liviana.

cargar_deptos <- function() {
  local <- file.path(carpeta_datos, "deptos_caba.csv")
  if (!file.exists(local)) {
    url <- "https://raw.githubusercontent.com/martintinch0/CienciaDeDatosParaCuriosos/master/data/datosProperati.csv"
    tmp <- tempfile(fileext = ".csv")
    download.file(url, tmp, mode = "wb")
    read_delim(tmp, delim = ";", locale = locale(encoding = "latin1"),
               show_col_types = FALSE) |>
      filter(property_type == "Departamento", operation_type == "Venta",
             currency == "USD") |>
      transmute(fecha     = as.Date(created_on),
                barrio    = str_to_title(BARRIO),
                ambientes = rooms,
                banios    = bathrooms,
                m2_total  = surface_total,
                m2        = surface_covered,
                precio    = price) |>
      write_csv(local)
    unlink(tmp)
  }
  read_csv(local, show_col_types = FALSE)
}

# Filtros de la clase: superficie cubierta entre 20 y 500 m2, precio entre
# 20 mil y 3 millones de dolares, hasta 8 ambientes, y solo barrios con al
# menos 300 avisos. Balvanera queda como barrio de referencia.
preparar_deptos <- function(deptos) {
  deptos |>
    filter(!is.na(m2), !is.na(ambientes), between(m2, 20, 500),
           between(precio, 20000, 3e6), ambientes <= 8) |>
    group_by(barrio) |>
    filter(n() >= 300) |>
    ungroup() |>
    mutate(barrio = relevel(factor(barrio), ref = "Balvanera"))
}

  }

# Ocupados con ingreso de la ocupacion principal (P21) positivo, con variables
# recodificadas. PONDIIO es el ponderador que corresponde a P21.
preparar_ocupados <- function(eph) {
  eph |>
    mutate(across(c(ESTADO, CAT_OCUP, P21, PONDIIO, PP3E_TOT, REGION,
                    NIVEL_ED, CH04, CH06), as.numeric)) |>
    filter(ESTADO == 1, P21 > 0) |>
    transmute(
      ingreso = P21,
      pondera = PONDIIO,
      horas   = if_else(PP3E_TOT > 0 & PP3E_TOT < 999, PP3E_TOT, NA_real_),
      edad    = CH06,
      sexo    = if_else(CH04 == 1, "Varon", "Mujer"),
      region  = factor(REGION, c(1, 40, 41, 42, 43, 44),
                       c("GBA", "NOA", "NEA", "Cuyo", "Pampeana", "Patagonia")),
      educacion = factor(case_when(NIVEL_ED %in% c(1, 2, 7) ~ "Hasta primario",
                                   NIVEL_ED == 3 ~ "Sec. incompleto",
                                   NIVEL_ED == 4 ~ "Sec. completo",
                                   NIVEL_ED == 5 ~ "Sup. incompleto",
                                   NIVEL_ED == 6 ~ "Sup. completo"),
                         c("Hasta primario", "Sec. incompleto", "Sec. completo",
                           "Sup. incompleto", "Sup. completo"))) |>
    filter(!is.na(educacion))
}

Parte 1. Regresión lineal con la EPH

Pregunta: ¿cuánto más gana por hora una persona con más educación, a igual sexo, edad y región?

El recorrido es siempre el mismo:

  1. Mirar los datos
  2. Estimar con lm()
  3. Interpretar signo y magnitud de cada coeficiente
  4. Testear significatividad y supuestos
  5. Corregir lo que haga falta y volver a interpretar
  6. Reportar en una tabla y en palabras

1.1 Armar la base

Ocupados del 1er trimestre de 2026 con ingreso de la ocupación principal positivo (clase18_datos.R ya hizo ese filtro y recodificó educación, sexo y región). Para comparar personas que trabajan distinta cantidad de horas, usamos el ingreso horario. Nos quedamos con quienes tienen entre 18 y 70 años y trabajan al menos 5 horas semanales.

eph <- ocupados |>
  filter(!is.na(horas), horas >= 5, between(edad, 18, 70)) |>
  mutate(ingreso_hora = ingreso / (horas * 4.3))   # 4,3 semanas por mes

nrow(eph)
glimpse(eph)

1.2 Mirar los datos antes de estimar

Un summary(), un histograma de la variable dependiente y un gráfico contra la regresora principal ahorran muchos problemas: valores imposibles, códigos como 99 = “no sabe”, asimetrías fuertes, ceros que impiden usar logaritmo.

summary(select(eph, ingreso_hora, edad, horas))
count(eph, educacion)
sum(eph$ingreso_hora <= 0)    # ¿hay ceros o negativos? log() no los acepta
options(repr.plot.height = 3.85)

hist(log(eph$ingreso_hora), breaks = 60, main = "log(ingreso horario)", xlab = "", col = "grey80")
par(mfrow = c(1, 1))

💡 El ingreso es muy asimétrico: muchos valores bajos y una cola larga a la derecha. En logaritmo queda casi simétrico. Además, en log los coeficientes se leen como cambios porcentuales, que es como se piensa un ingreso. Por eso la variable dependiente va a ser log(ingreso_hora).

options(repr.plot.height = 3.85)
ggplot(eph, aes(educacion, log(ingreso_hora))) +
  geom_boxplot(fill = "grey90") +
  labs(x = NULL, y = "log(ingreso horario)")

1.3 Primer modelo: solo educación

educacion es un factor con 5 niveles. lm() arma solo las k − 1 = 4 dummies. La que falta es la categoría base (“Hasta primario”, el primer nivel), y todos los coeficientes se leen en comparación con ella.

e0 <- lm(log(ingreso_hora) ~ educacion, data = eph)
summary(e0)

Cómo se lee la salida de summary()

Parte Qué es Qué mirar
Estimate El coeficiente Signo y magnitud: es lo que se interpreta
Std. Error Error estándar Cuán precisa es la estimación
t value Coeficiente / EE Estadístico del test \(H_0: \beta = 0\)
Pr(>|t|) p-valor Si es menor a 0,05, el coeficiente es significativo al 5%
Residual standard error Desvío del error En unidades de \(y\) (acá, log-puntos)
Multiple R-squared \(R^2\) Proporción de la variabilidad de \(y\) que explica el modelo
Adjusted R-squared \(R^2\) ajustado Penaliza agregar variables; sirve para comparar modelos

Con solo dummies, la regresión reproduce las medias de cada grupo: el intercepto es la media de la base y cada coeficiente es la diferencia con ella.

eph |>
  group_by(educacion) |>
  summarise(media_log = mean(log(ingreso_hora))) |>
  mutate(diferencia_con_base = media_log - first(media_log))

Dummies con log(y): la diferencia exacta

Con la variable dependiente en log, el coeficiente × 100 es solo una aproximación del cambio porcentual, que empeora cuando el coeficiente es grande (más de 0,1 en valor absoluto). La diferencia exacta es

\[\%\Delta y = 100 \cdot \left(e^{\hat\beta} - 1\right)\]

b0 <- coef(e0)[-1]
tibble(nivel       = names(b0),
       beta        = b0,
       aproximado  = 100 * b0,
       exacto      = 100 * (exp(b0) - 1))

💡 Sin controlar por nada, quien tiene universitario completo gana por hora 106% más que quien tiene hasta primario (la lectura rápida, 72%, se queda muy corta).

1.4 Agregar controles: “manteniendo lo demás constante”

En una regresión múltiple, cada coeficiente es el efecto de su variable manteniendo constantes las demás incluidas en el modelo. Agregamos sexo, edad (con su cuadrado, ver 1.5) y región.

e1 <- lm(log(ingreso_hora) ~ educacion + sexo + edad + I(edad^2) + region, data = eph)
summary(e1)

¿Cambiaron los coeficientes de educación?

cbind(solo_educacion = coef(e0)[2:5], con_controles = coef(e1)[2:5])

💡 Los coeficientes de educación suben al controlar. Las personas más educadas son, en promedio, más jóvenes que las de la base (que tienen 48 años de media) y entre ellas hay más mujeres (58% entre universitarios completos, 33% en hasta primario); las mujeres ganan menos por hora. Sin esos controles, el modelo simple le restaba a la educación parte de su asociación con el ingreso. Es el sesgo por variable omitida: cuando un coeficiente cambia mucho al agregar un control, el control importaba.

1.5 Interpretar cada coeficiente

b1 <- coef(e1)
round(100 * (exp(b1[c(2:6, 9:13)]) - 1), 1)    # diferencias % exactas de cada dummy

💡 - Educación: a igual sexo, edad y región, universitario completo gana por hora 118% más que hasta primario; secundario completo, 31% más. - Sexo (base: Mujer): a igual educación, edad y región, los varones ganan por hora 20% más. No alcanza para hablar de discriminación: faltan ocupación, sector, formalidad. - Región (base: GBA): a iguales características, en el NEA se gana 39% menos por hora y en la Patagonia, 20% más.

La edad con un término cuadrático

El efecto de un año más de edad no es el mismo a los 25 que a los 55. Con I(edad^2) (el I() es obligatorio dentro de una fórmula), el efecto marginal depende de la edad:

\[\frac{\partial \log(y)}{\partial\, \text{edad}} = \hat\beta_{edad} + 2\,\hat\beta_{edad^2}\cdot \text{edad}\]

El coeficiente de edad solo no alcanza para interpretar.

efecto_edad <- function(x) 100 * (b1["edad"] + 2 * b1["I(edad^2)"] * x)
c(a_los_25 = efecto_edad(25), a_los_40 = efecto_edad(40), a_los_55 = efecto_edad(55))

punto_giro <- -b1["edad"] / (2 * b1["I(edad^2)"])
punto_giro
mean(eph$edad > punto_giro)     # qué parte de la muestra está después del máximo

💡 A los 25 años, un año más se asocia con un ingreso horario 2,1% mayor; a los 40, 1%; cerca de los 54 años el perfil llega a su máximo. Antes de decir que “después de los 54 el ingreso cae”, mirar cuántas observaciones hay más allá de ese punto.

options(repr.plot.height = 3.85)
grilla <- tibble(edad = 18:70, educacion = "Sec. completo", sexo = "Mujer", region = "GBA")
grilla$pred <- predict(e1, newdata = grilla)
ggplot(grilla, aes(edad, exp(pred))) +
  geom_line(linewidth = 1, color = "firebrick") +

Cambiar la categoría base

relevel() cambia la base. Cambian los coeficientes, no las predicciones.

eph_b <- eph |> mutate(region = relevel(region, ref = "Pampeana"))
e1_b  <- update(e1, data = eph_b)
coef(e1_b)[str_detect(names(coef(e1_b)), "region")]
all.equal(fitted(e1), fitted(e1_b))

⚠️ Nunca meter una categórica codificada como número. En la EPH, REGION viene como 1, 40, 41, 42, 43, 44. Si entra así, R la trata como continua y estima un único efecto por “subir un código”, que no significa nada:

eph_mal <- eph |> mutate(region_codigo = c(1, 40, 41, 42, 43, 44)[as.integer(region)])
coef(lm(log(ingreso_hora) ~ educacion + sexo + edad + I(edad^2) + region_codigo,
        data = eph_mal))["region_codigo"]

1.6 Interacciones

¿La brecha por sexo es la misma en todos los niveles educativos? Una interacción permite que el efecto de una variable cambie según otra. educacion * sexo incluye educacion, sexo y los productos educacion:sexo.

e2 <- lm(log(ingreso_hora) ~ educacion * sexo + edad + I(edad^2) + region, data = eph)
coeftest(e2)
  • sexoVaron: brecha por sexo en la base (hasta primario).
  • educacionX:sexoVaron: cuánto cambia esa brecha en el nivel X.
  • Brecha en el nivel X = sexoVaron + educacionX:sexoVaron.
b2 <- coef(e2)
niveles <- levels(eph$educacion)
brecha <- c(b2["sexoVaron"],
            b2["sexoVaron"] + b2[paste0("educacion", niveles[-1], ":sexoVaron")])
tibble(educacion = niveles, brecha_exacta_pct = 100 * (exp(brecha) - 1))

anova(e1, e2)    # ¿la interacción importa en conjunto?

💡 La brecha a favor de los varones es de 26% entre quienes tienen hasta primario y baja a 13% entre universitarios completos. El test F conjunto rechaza al 5% (p = 0,02): la interacción aporta.

1.7 Significatividad

  • Cada coeficiente: test \(t\), \(H_0: \beta_j = 0\). Equivalente: si el intervalo de confianza no contiene al cero, el coeficiente es significativo.
  • Varios a la vez: test \(F\). Caso típico: todas las dummies de una categórica, o un término y su cuadrado.
coeftest(e1)
confint(e1)

e1_sin_region <- lm(log(ingreso_hora) ~ educacion + sexo + edad + I(edad^2), data = eph)
anova(e1_sin_region, e1)     # H0: las 5 dummies de región son cero a la vez

⚠️ Significativo no es lo mismo que importante. Con casi 14 mil observaciones, efectos chicos salen significativos. Mirar siempre la magnitud y preguntarse si es económicamente relevante.

1.8 Diagnóstico: chequear los supuestos y corregir

Problema Qué rompe Test en R Solución
Forma funcional incorrecta Coeficientes (sesgo) resettest() Logs, cuadrados, interacciones
Multicolinealidad Nada, pero infla los EE vif() Quitar o combinar variables

💡 La lógica de todos los tests es la misma: \(H_0\) = “no hay problema”. p-valor < 0,05 → se rechaza → hay evidencia del problema. Con muestras grandes casi todo test rechaza: mirar también los gráficos antes de decidir.

Primera mirada: los gráficos de plot()

options(repr.plot.height = 7.7)

Forma funcional: test RESET

Agrega potencias de los valores predichos (\(\hat y^2, \hat y^3\)) y testea si aportan algo. Si aportan, falta alguna no linealidad. El test no dice qué falta.

resettest(e1)
resettest(e2)

💡 Rechaza, aunque con un estadístico chico para 14 mil observaciones. Ya tenemos logaritmo, el cuadrado de la edad y la interacción; el resto de la no linealidad probablemente venga de variables que no están (ocupación, formalidad). Qué hacer si rechaza: probar logs, cuadrados o interacciones con sentido económico.

Multicolinealidad: VIF

No genera sesgo: con regresoras muy correlacionadas entre sí, los errores estándar se inflan. Regla práctica: VIF > 10 merece atención.

vif(e1)

💡 edad y edad^2 tienen VIF altísimo por construcción (son la misma variable): no es un problema. El resto está cerca de 1. Qué hacer en general: si las variables colineales son controles, nada. Si importa separar sus efectos, quitar una redundante o combinarlas en un índice. Nunca sacar una variable relevante solo para bajar el VIF: se genera sesgo por omisión.

Heteroscedasticidad: detectar

La varianza del error cambia con las regresoras. Los coeficientes siguen bien, pero los errores estándar están mal, y con ellos los tests y los intervalos.

options(repr.plot.height = 3.85)
plot(fitted(e1), residuals(e1), pch = 16, col = rgb(0, 0, 1, 0.1),
     xlab = "Valores predichos", ylab = "Residuos")
abline(h = 0, lty = 2)

Test de Breusch-Pagan: regresa los residuos al cuadrado contra las regresoras. \(H_0\): varianza constante.

bptest(e1)

Heteroscedasticidad: corregir con errores robustos

Errores estándar robustos de White: mismos coeficientes, errores estándar recalculados con una fórmula válida con o sin heteroscedasticidad. HC1 es la misma corrección que usa Stata.

coeftest(e1)                                        # errores clásicos
coeftest(e1, vcov = vcovHC(e1, type = "HC1"))       # errores robustos
tibble(termino    = names(coef(e1)),
       coef       = coef(e1),
       ee_clasico = sqrt(diag(vcov(e1))),
       ee_robusto = sqrt(diag(vcovHC(e1, type = "HC1")))) |>
  mutate(cociente = ee_robusto / ee_clasico)

Con errores robustos también se hacen los intervalos y los tests conjuntos:

coefci(e1, vcov. = vcovHC(e1, type = "HC1"))
waldtest(e1_sin_region, e1, vcov = vcovHC(e1, type = "HC1"))

💡 Hay heteroscedasticidad, pero acá la corrección cambia poco los errores estándar (a lo sumo un 7%). En corte transversal se reportan casi siempre errores robustos: si no hay heteroscedasticidad dan casi lo mismo que los clásicos; si la hay, son los correctos.

Parte 2. Shift-share

La pregunta

Cuando cambia un agregado que es un promedio ponderado, ¿cuánto vino de cambios dentro de cada componente y cuánto de la recomposición entre componentes?

  • Productividad de un país: promedio de la productividad de cada sector, ponderada por empleo.
  • Margen de una empresa: promedio del margen de cada producto, ponderado por ventas.
  • Salario medio: promedio por ocupación, ponderado por cantidad de ocupados.

El agregado puede caer aunque todos los componentes mejoren, si el peso se mueve hacia los de nivel más bajo (paradoja de Simpson). El shift-share mide exactamente cuánto de cada cosa.

La identidad y su descomposición

La productividad agregada es el producto total sobre el empleo total. Escrita como promedio ponderado, con \(s_i = L_i / L\) la participación del sector \(i\) en el empleo y \(p_i = Y_i / L_i\) su productividad:

\[P = \frac{\sum_i Y_i}{\sum_i L_i} = \sum_i s_i \, p_i\]

Entre el año 0 y el año 1, \(s_{i1} = s_{i0} + \Delta s_i\) y \(p_{i1} = p_{i0} + \Delta p_i\). Para cada sector:

\[s_{i1}\,p_{i1} - s_{i0}\,p_{i0} = (s_{i0} + \Delta s_i)(p_{i0} + \Delta p_i) - s_{i0}\,p_{i0} = s_{i0}\,\Delta p_i + \Delta s_i\, p_{i0} + \Delta s_i\, \Delta p_i\]

Sumando sobre los sectores:

\[\Delta P = \underbrace{\sum_i s_{i0}\,\Delta p_i}_{\text{intrasectorial}} + \underbrace{\sum_i \Delta s_i \, p_{i0}}_{\text{estático}} + \underbrace{\sum_i \Delta s_i \, \Delta p_i}_{\text{dinámico}}\]

  • Intrasectorial: mejoras dentro de cada sector, con la estructura del año inicial.
  • Estático: reasignación del empleo hacia sectores que ya eran más productivos.
  • Dinámico: positivo si el empleo va hacia sectores cuya productividad crece.

Es pura álgebra: los tres términos suman exacto el cambio total. No hay nada estimado, no hay errores estándar.

shift_share <- function(d) {
  d |> mutate(intra    = s0 * (p1 - p0),
              estatico = (s1 - s0) * p0,
              dinamico = (s1 - s0) * (p1 - p0))
}

Ejemplo de pizarrón

El problema del año base

El término dinámico es un “sobrante” que no pertenece naturalmente a ninguno de los otros dos. Hay varias formas de repartirlo, todas exactas:

Variante Dentro de sectores Reasignación
Tres términos \(\sum s_{i0}\Delta p_i\) \(\sum \Delta s_i\, p_{i0}\) y \(\sum \Delta s_i \Delta p_i\) por separado
Pesos finales en el intra \(\sum s_{i1}\Delta p_i\) \(\sum \Delta s_i\, p_{i0}\)
Productividad final en la reasignación (McMillan y Rodrik) \(\sum s_{i0}\Delta p_i\) \(\sum \Delta s_i\, p_{i1}\)
Promedios de los dos años \(\sum \bar s_i \Delta p_i\) \(\sum \Delta s_i\, \bar p_i\)

Es el mismo problema de los números índice (Laspeyres vs Paasche).

En negocios: el margen de Superstore (precio-volumen-mix)

El margen total es un promedio de los márgenes de cada producto, ponderado por su participación en las ventas:

\[M = \frac{\sum_i G_i}{\sum_i V_i} = \sum_i w_i \, m_i \qquad w_i = \frac{V_i}{V}, \quad m_i = \frac{G_i}{V_i}\]

¿Cambió el margen porque cada producto rinde distinto o porque cambió qué se vende? En finanzas corporativas es el análisis precio-volumen-mix: el término estático es el “efecto mix”. Se usa la misma función, con \(s\) = participación en ventas y \(p\) = margen.

Versión regional o por sucursal

Para el cambio de las ventas (o del empleo) de una región \(r\) en el sector \(i\), con \(g\) = tasa de crecimiento entre los dos años:

\[\Delta e_{ri} = e_{ri0}\, g_{ri}\]

Sumando y restando el crecimiento total (\(g_N\)) y el del sector en el total (\(g_{Ni}\)):

\[\Delta e_{ri} = \underbrace{e_{ri0}\, g_N}_{\text{nacional}} + \underbrace{e_{ri0}\,(g_{Ni} - g_N)}_{\text{composición}} + \underbrace{e_{ri0}\,(g_{ri} - g_{Ni})}_{\text{competitivo}}\]

  • Nacional o de cadena: lo que habría crecido al ritmo del total.
  • Composición: lo que aporta estar especializado en sectores o categorías dinámicas.
  • Competitivo: desempeño propio respecto del mismo sector en el total.
reg <- superstore |>
  filter(anio %in% c(2015, 2018)) |>
  group_by(region, categoria, anio) |>
  summarise(v = sum(ventas), .groups = "drop") |>

  group_by(categoria) |>

         nacional    = v2015 * g_N,
         composicion = v2015 * (g_Ni - g_N),
         competitivo = v2015 * (g_ri - g_Ni)) |>

Cuidados

  • El resultado depende del año base y de cómo se reparte el término cruzado.
  • El nivel de agregación cambia el reparto: lo que con 9 sectores es “intrasectorial” puede ser reasignación entre ramas con 50 sectores.
  • Unidades comparables: productividad a precios constantes. Con precios corrientes, el cambio de precios relativos se mezcla con la productividad.
  • Es una identidad: describe dónde está el cambio y deja abierta la pregunta por su causa.

Parte 3. Entropía e índice de Theil

Antes: qué es la entropía

Viene de la teoría de la información (Shannon, 1948). Si un evento tiene probabilidad \(p\), la información (o sorpresa) que aporta enterarse de que ocurrió es

\[h(p) = -\ln p\]

Un evento seguro (\(p = 1\)) no sorprende: \(h = 0\). Uno rarísimo sorprende mucho. La entropía es la sorpresa esperada de una distribución \((p_1, \dots, p_N)\):

\[H = -\sum_i p_i \ln p_i\]

  • \(H = 0\) si un resultado tiene toda la probabilidad: no hay incertidumbre.
  • \(H = \ln N\), su máximo, si todos los resultados son igual de probables.
  • Mide cuán “repartida” está una distribución.
entropia <- function(p) { p <- p / sum(p); -sum(p * log(p)) }

entropia(c(0.5, 0.5))      # moneda justa: ln 2
entropia(c(0.99, 0.01))    # moneda cargada: casi no hay nada que adivinar
entropia(rep(1, 6))        # dado: ln 6
log(6)

De la entropía a la desigualdad

Theil (1967) reemplazó probabilidades por participaciones en el ingreso, \(q_i = y_i / \sum_j y_j\):

  • Si todos ganan lo mismo, \(q_i = 1/N\) y la entropía es máxima: \(\ln N\).
  • Si una persona se queda con todo, la entropía es 0.

La desigualdad es lo que le falta a la entropía para llegar al máximo:

\[T = \ln N - H(q) = \ln N + \sum_i q_i \ln q_i = \sum_i q_i \ln (N q_i)\]

Como \(q_i = \dfrac{y_i}{N \bar y}\), reemplazando se llega a la fórmula usual:

\[T = \frac{1}{N} \sum_i \frac{y_i}{\bar{y}} \ln\left(\frac{y_i}{\bar{y}}\right)\]

Otra lectura: \(T = \sum_i q_i \ln\left(\dfrac{q_i}{1/N}\right)\) es la divergencia de Kullback-Leibler entre la distribución del ingreso (\(q_i\)) y la de la población (\(1/N\)). Mide qué tan distinto es “quién tiene el ingreso” de “quién es la gente”.

theil <- function(y, w = rep(1, length(y))) {
  mu <- weighted.mean(y, w)
  sum(w * (y / mu) * log(y / mu)) / sum(w)
}

y <- c(10, 20, 30, 40)
c(theil = theil(y), lnN_menos_H = log(length(y)) - entropia(y))   # iguales

theil(c(25, 25, 25, 25))           # igualdad perfecta: 0
theil(c(0.001, 0.001, 0.001, 100)) # casi todo en una persona...
log(4)                             # ...se acerca al máximo, ln N

Propiedades

Un buen índice de desigualdad cumple:

  • Independencia de escala: multiplicar todos los ingresos por una constante no cambia el índice.
  • Principio de población: replicar la población no cambia el índice.
  • Transferencias (Pigou-Dalton): pasar ingreso de alguien más rico a alguien más pobre baja el índice.
theil(y * 1000)              # escala
theil(rep(y, 3))             # población
theil(c(15, 20, 30, 35))     # transferencia de 5 del más rico al más pobre: baja

El Theil T es parte de la familia de entropía generalizada:

\[GE(\alpha) = \frac{1}{\alpha(\alpha-1)} \frac{1}{N}\sum_i \left[\left(\frac{y_i}{\bar y}\right)^\alpha - 1\right]\]

  • \(\alpha = 0\): desvío logarítmico medio (Theil L), más sensible a la parte baja de la distribución.
  • \(\alpha = 1\): Theil T.
  • \(\alpha = 2\): la mitad del cuadrado del coeficiente de variación, sensible a la parte alta.

Desigualdad del ingreso laboral (EPH 1T2026)

Con microdatos de encuesta hay que usar el ponderador (PONDIIO para el ingreso de la ocupación principal). Sin pesos se calcula la desigualdad de la muestra, no la de la población.

mld <- function(y, w = rep(1, length(y))) {
  mu <- weighted.mean(y, w)
  sum(w * log(mu / y)) / sum(w)
}
gini <- function(y, w = rep(1, length(y))) {
  o <- order(y); y <- y[o]; w <- w[o]
  pw <- w / sum(w); L <- cumsum(w * y) / sum(w * y)
  1 - sum(pw * (L + c(0, head(L, -1))))
}

c(theil_T = theil(ocupados$ingreso, ocupados$pondera),
  theil_L = mld(ocupados$ingreso, ocupados$pondera),
  gini    = gini(ocupados$ingreso, ocupados$pondera),
  theil_T_sin_pesos = theil(ocupados$ingreso))

Descomposición entre y dentro de grupos

Con \(s_g\) la participación del grupo \(g\) en el ingreso total, \(T_g\) el Theil dentro del grupo y \(\bar y_g\) su media:

\[T = \underbrace{\sum_g s_g \, T_g}_{\text{dentro}} + \underbrace{\sum_g s_g \ln\left(\frac{\bar{y}_g}{\bar{y}}\right)}_{\text{entre}}\]

  • Dentro: desigualdad al interior de cada grupo, ponderada por el peso del grupo en el ingreso.
  • Entre: la desigualdad que quedaría si todos en un grupo ganaran la media del grupo.

Por qué es exacta: escribiendo \(\dfrac{y_i}{\bar y} = \dfrac{y_i}{\bar y_g} \cdot \dfrac{\bar y_g}{\bar y}\), el logaritmo del producto se separa en una suma: un término con la distancia de cada persona a la media de su grupo y otro con la distancia entre la media del grupo y la general.

El Gini no tiene una descomposición exacta así (queda un residuo cuando los grupos se superponen): por eso para esta pregunta se usa Theil.

theil_desc <- function(d, grupo) {
  mu <- weighted.mean(d$ingreso, d$pondera)
  d |>
    group_by(grupo = {{ grupo }}) |>
    summarise(pob   = sum(pondera),
              media = weighted.mean(ingreso, pondera),
              T_g   = theil(ingreso, pondera),
              .groups = "drop") |>
    mutate(s_g    = pob * media / sum(pob * media),
           dentro = s_g * T_g,
           entre  = s_g * log(media / mu))
}

por_educ <- theil_desc(ocupados, educacion)
por_educ

por_educ |>
  summarise(dentro = sum(dentro), entre = sum(entre)) |>
  mutate(total       = dentro + entre,
         chequeo     = theil(ocupados$ingreso, ocupados$pondera),
         share_entre = entre / total)

⚠️ Cuidados

  • Hay que filtrar ceros y negativos (\(\ln 0\) no existe).
  • El Theil T es más sensible a lo que pasa en la parte alta de la distribución.
  • El “entre” crece mecánicamente con la cantidad de grupos. Educación × región tiene 30 grupos:
theil_desc(ocupados, interaction(educacion, region)) |>
  summarise(entre = sum(entre)) |>
  mutate(share_entre = entre / theil(ocupados$ingreso, ocupados$pondera))
